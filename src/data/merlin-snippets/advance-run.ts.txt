/**
 * Walk a run through its nodes until it has to wait or it ends. The run holds a 3-minute lock:
 * the scheduler tick and a start for a new lead never take it at the same time.
 */
export async function advanceRun(
  runId: string,
): Promise<"progressed" | "idle" | "busy" | "failed"> {
  const [claimed] = await db
    .update(merlinSkillRuns)
    .set({ lockedUntil: new Date(Date.now() + LOCK_MS) })
    .where(
      and(
        eq(merlinSkillRuns.id, runId),
        inArray(merlinSkillRuns.status, ["running", "waiting"]),
        sql`(${merlinSkillRuns.lockedUntil} is null or ${merlinSkillRuns.lockedUntil} < now())`,
      ),
    )
    .returning();
  if (!claimed) return "busy";

        return "progressed";
      }
      const result = await execNode(node, { cur, skill, lead });
      if (result.kind === "wait") {
        cur.status = "waiting";
        cur.waitUntil = result.until;
        cur.waitReply = result.reply;
        if (result.since !== undefined) cur.waitSince = result.since;
        await save(cur);

        await finish("failed", result.code, result.error);
